# Kerf v2 Core (3D) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A 3D signed-distance ray-marching renderer (shapes, perspective/ortho cameras, sun with soft shadows, sky with AO, emission, HDR + filmic present) whose tile-incremental output is byte-identical to its brute-force reference, plus a first frame-time read.

**Architecture:** New engine in `engine/kerf2/` beside v1 (v1 stays until Plan B migrates the arena). Same three compute passes as v1 — `clear`, `bin`, `shade` — over 64-byte shape records. `bin` computes each shape's *screen reach* (projection of its bounding box grown by the step cap + AO reach, swept along the sun direction to the floor) and marks reached tiles; `shade` sphere-traces each pixel of a dirty tile against only that tile's shapes, with every distance clamped by the step cap `D` so excluded shapes can never change a step. Output is linear HDR in an `rgba16float` tile cache; a present pass tone-maps to the canvas.

**Tech Stack:** TypeScript, WGSL/WebGPU, Vite 6, Vitest 3 (node + Chrome browser mode), Playwright (bench).

**Spec:** `docs/superpowers/specs/2026-10-08-kerf-v2-design.md` (this plan covers §2, §3.1, §3.3, §4.1, §4.3 sun + sky, §4.4, §5, §6, §8 equivalence/bench. Plan B covers §3.2 groups, §4.2 materials + jelly, §4.3 player lights + haze, v1 2D compatibility, arena migration, head-to-head re-run.)

## Global Constraints

- WebGPU only; no Three.js or runtime framework.
- y up, floor plane `y = 0`, arena 1600 × 900 on x/z.
- Shape record 64 bytes; max 2048 shapes; exceeding throws; non-finite values throw.
- Every march step advances `min(d, D)` with the same `D` in fast and brute modes.
- Fast output byte-identical to brute on the HDR cache; a difference is a bug, never a tolerance.
- Camera, sun or sky change ⇒ full redraw. Exposure lives in the present pass and never dirties tiles.
- Perf gate (reported, not enforced in this plan): worst-case p99 ≤ 16.6 ms at 1080p and 4K.

## Review Focus

1. A shape straddling or behind the camera's near plane must still render correctly (reach = whole screen) → test in Task 3.
2. A moving shape's shadow can land far from the shape on screen; those tiles must redraw (sun sweep) → test in Task 3 (tall pillar, low sun).
3. A long thin ellipsoid near a tile edge (lower-bound distance far along its axis) must not change steps in tiles that exclude it → test in Task 3.
4. Sun or camera change must redraw everything → test in Task 3.
5. Widths not divisible by 16/32 (1366) must shade partial tiles and read back HDR (8 bytes/px) without row skew → test in Task 2.

```ts
// ponytail: 2048-shape cap keeps a tile's list in workgroup memory (same as v1).
export const MAX_SHAPES = 2048;
export const SHAPE_BYTES = 64;
export const SHAPE_FLOATS = 16;
export const Kind = { None: 0, Sphere: 1, Ellipsoid: 2, Capsule: 3, RoundBox: 4, Cylinder: 5, Plane: 6 } as const;

export type Quat = readonly [number, number, number, number];
export const IDENTITY: Quat = [0, 0, 0, 1];
export interface ShapeOpts { material?: number; group?: number; emission?: number }

export function rgba(r: number, g: number, b: number, a = 255): number {
  return ((a << 24) | (b << 16) | (g << 8) | r) >>> 0;
}

export class ShapeList {
  readonly data = new ArrayBuffer(MAX_SHAPES * SHAPE_BYTES);
  private f = new Float32Array(this.data);
  private u = new Uint32Array(this.data);
  count = 0;

  clear(): void {
    this.count = 0;
  }

  sphere(x: number, y: number, z: number, r: number, color: number, o: ShapeOpts = {}): void {
    this.push(Kind.Sphere, x, y, z, r, 0, 0, 0, IDENTITY, color, o);
  }

  ellipsoid(x: number, y: number, z: number, ax: number, ay: number, az: number, rot: Quat, color: number, o: ShapeOpts = {}): void {
    this.push(Kind.Ellipsoid, x, y, z, ax, ay, az, 0, rot, color, o);
  }

  capsule(ax: number, ay: number, az: number, bx: number, by: number, bz: number, r: number, color: number, o: ShapeOpts = {}): void {
    this.push(Kind.Capsule, ax, ay, az, bx - ax, by - ay, bz - az, r, IDENTITY, color, o);
  }

  roundbox(x: number, y: number, z: number, hx: number, hy: number, hz: number, round: number, rot: Quat, color: number, o: ShapeOpts = {}): void {
    this.push(Kind.RoundBox, x, y, z, hx, hy, hz, round, rot, color, o);
  }

  cylinder(x: number, y: number, z: number, r: number, halfH: number, round: number, rot: Quat, color: number, o: ShapeOpts = {}): void {
    this.push(Kind.Cylinder, x, y, z, r, halfH, 0, round, rot, color, o);
  }

  plane(y: number, color: number, o: ShapeOpts = {}): void {
    this.push(Kind.Plane, 0, y, 0, 0, 0, 0, 0, IDENTITY, color, o);
  }

  private push(kind: number, px: number, py: number, pz: number, sx: number, sy: number, sz: number, round: number, q: Quat, color: number, o: ShapeOpts): void {
    if (this.count >= MAX_SHAPES) throw new Error(`Kerf: shape limit ${MAX_SHAPES} exceeded`);
    const ql = Math.hypot(q[0], q[1], q[2], q[3]);
    const em = o.emission ?? 0;
    if (!Number.isFinite(px + py + pz + sx + sy + sz + round + ql + em) || ql === 0) {
      throw new Error('Kerf: shape values must be finite');
    }
    const i = this.count++ * SHAPE_FLOATS;
    const f = this.f;
    f[i] = px; f[i + 1] = py; f[i + 2] = pz;
    this.u[i + 3] = kind;
    f[i + 4] = sx; f[i + 5] = sy; f[i + 6] = sz; f[i + 7] = round;
    f[i + 8] = q[0] / ql; f[i + 9] = q[1] / ql; f[i + 10] = q[2] / ql; f[i + 11] = q[3] / ql;
    this.u[i + 12] = color >>> 0;
    this.u[i + 13] = o.material ?? 0;
    this.u[i + 14] = o.group ?? 0;
    f[i + 15] = em;
  }
}
```

```ts
export function f16(bits: number): number {
  const s = bits & 0x8000 ? -1 : 1;
  const e = (bits >> 10) & 0x1f;
  const m = bits & 0x3ff;
  if (e === 0) return s * m * 2 ** -24;
  if (e === 31) return m ? NaN : s * Infinity;
  return s * (1 + m / 1024) * 2 ** (e - 15);
}

export function hdrPixel(bytes: Uint8Array, width: number, x: number, y: number): [number, number, number, number] {
  const v = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
  const o = (y * width + x) * 8;
  return [f16(v.getUint16(o, true)), f16(v.getUint16(o + 2, true)), f16(v.getUint16(o + 4, true)), f16(v.getUint16(o + 6, true))];
}
```

```wgsl
const TILE: u32 = 16u;
const MAX_SHAPES: u32 = 2048u;
const MASK_WORDS: u32 = 64u;
const KIND_NONE: u32 = 0u;
const KIND_SPHERE: u32 = 1u;
const KIND_ELLIPSOID: u32 = 2u;
const KIND_CAPSULE: u32 = 3u;
const KIND_ROUNDBOX: u32 = 4u;
const KIND_CYLINDER: u32 = 5u;
const KIND_PLANE: u32 = 6u;

// Exactness (spec §5): every distance used for stepping is min(d, STEP_CAP); a tile's list holds every
// shape whose bounds grown by REACH_GROW (= STEP_CAP + AO_MAX) reach its rays, so excluded shapes are
// always >= STEP_CAP away at every point any march or AO sample visits.
const STEP_CAP: f32 = 48.0;
const AO_MAX: f32 = 24.0;
const REACH_GROW: f32 = 72.0;
const SWEEP_FLOOR: f32 = -72.0;
const MAX_STEPS: u32 = 160u;
const MAX_DIST: f32 = 8000.0;
const SHADOW_STEPS: u32 = 48u;
const SHADOW_MAX: f32 = 1500.0;
const SURF_EPS: f32 = 0.05;
const PAD_PX: f32 = 2.0;

struct Shape { pos: vec3f, kind: u32, size: vec3f, round: f32, rot: vec4f, color: u32, material: u32, group: u32, emission: f32 }
struct Cam { eye: vec3f, mode: u32, right: vec3f, sx: f32, up: vec3f, sy: f32, fwd: vec3f, near: f32 }
struct Env { sunDir: vec3f, sunAngle: f32, sunColor: vec3f, sunIntensity: f32, skyColor: vec3f, skyIntensity: f32 }
struct Params { count: u32, prevCount: u32, tilesX: u32, tilesY: u32, forceAll: u32, brute: u32, width: u32, height: u32 }
struct Uni { cam: Cam, env: Env, p: Params }

@group(0) @binding(0) var<uniform> U: Uni;
@group(0) @binding(1) var<storage, read> shapes: array<Shape>;
@group(0) @binding(2) var<storage, read> prevShapes: array<Shape>;
@group(0) @binding(3) var<storage, read_write> tileMasks: array<atomic<u32>>;
@group(0) @binding(4) var<storage, read_write> dirty: array<atomic<u32>>;
@group(0) @binding(5) var hdr: texture_storage_2d<rgba16float, write>;

// ---------- clear ----------
@compute @workgroup_size(256)
fn clear(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  let tiles = U.p.tilesX * U.p.tilesY;
  if (i < tiles * MASK_WORDS) { atomicStore(&tileMasks[i], 0u); }
  if (i < (tiles + 31u) / 32u) { atomicStore(&dirty[i], select(0u, 0xffffffffu, U.p.forceAll != 0u)); }
}

// ---------- geometry ----------
fn qrot(q: vec4f, v: vec3f) -> vec3f {
  let t = 2.0 * cross(q.xyz, v);
  return v + q.w * t + cross(q.xyz, t);
}

fn sdShape(pw: vec3f, s: Shape) -> f32 {
  if (s.kind == KIND_PLANE) { return pw.y - s.pos.y; }
  if (s.kind == KIND_CAPSULE) {
    let pa = pw - s.pos;
    let h = clamp(dot(pa, s.size) / max(dot(s.size, s.size), 1e-12), 0.0, 1.0);
    return length(pa - s.size * h) - s.round;
  }
  let p = qrot(vec4f(-s.rot.xyz, s.rot.w), pw - s.pos);
  switch s.kind {
    case KIND_SPHERE: { return length(p) - s.size.x; }
    case KIND_ELLIPSOID: {
      let r = s.size;
      // lower bound with the true zero set; second term keeps it >= distance to the bounding sphere (spec §5)
      return max((length(p / r) - 1.0) * min(r.x, min(r.y, r.z)), length(p) - max(r.x, max(r.y, r.z)));
    }
    case KIND_ROUNDBOX: {
      let q = abs(p) - s.size + vec3f(s.round);
      return length(max(q, vec3f(0.0))) + min(max(q.x, max(q.y, q.z)), 0.0) - s.round;
    }
    default: { // cylinder along local y
      let d = vec2f(length(p.xz), abs(p.y)) - vec2f(s.size.x, s.size.y) + vec2f(s.round);
      return min(max(d.x, d.y), 0.0) + length(max(d, vec2f(0.0))) - s.round;
    }
  }
}

// Bounding sphere (center, radius); radius < 0 = unbounded (plane).
fn bound(s: Shape) -> vec4f {
  switch s.kind {
    case KIND_SPHERE: { return vec4f(s.pos, s.size.x); }
    case KIND_ELLIPSOID: { return vec4f(s.pos, max(s.size.x, max(s.size.y, s.size.z))); }
    case KIND_CAPSULE: { return vec4f(s.pos + s.size * 0.5, length(s.size) * 0.5 + s.round); }
    case KIND_ROUNDBOX: { return vec4f(s.pos, length(s.size)); }
    case KIND_CYLINDER: { return vec4f(s.pos, length(s.size.xy)); }
    default: { return vec4f(s.pos, -1.0); }
  }
}

// ---------- camera ----------
fn ndcToPixel(n: vec2f) -> vec2f {
  return vec2f((n.x + 1.0) * 0.5 * f32(U.p.width), (1.0 - n.y) * 0.5 * f32(U.p.height));
}

// World point -> pixel. Sets *ok = false if the point is behind the camera plane.
fn project(w: vec3f, ok: ptr<function, bool>) -> vec2f {
  let v = w - U.cam.eye;
  let z = dot(v, U.cam.fwd);
  if (U.cam.mode == 0u) {
    if (z < U.cam.near) { *ok = false; return vec2f(0.0); }
    return ndcToPixel(vec2f(dot(v, U.cam.right) / (z * U.cam.sx), dot(v, U.cam.up) / (z * U.cam.sy)));
  }
  if (z < 0.0) { *ok = false; return vec2f(0.0); }
  return ndcToPixel(vec2f(dot(v, U.cam.right) / U.cam.sx, dot(v, U.cam.up) / U.cam.sy));
}

struct Ray { o: vec3f, d: vec3f }

fn pixelRay(pix: vec2u) -> Ray {
  let n = vec2f((f32(pix.x) + 0.5) / f32(U.p.width) * 2.0 - 1.0, 1.0 - (f32(pix.y) + 0.5) / f32(U.p.height) * 2.0);
  if (U.cam.mode == 0u) {
    return Ray(U.cam.eye, normalize(U.cam.fwd + n.x * U.cam.sx * U.cam.right + n.y * U.cam.sy * U.cam.up));
  }
  return Ray(U.cam.eye + n.x * U.cam.sx * U.cam.right + n.y * U.cam.sy * U.cam.up, U.cam.fwd);
}

// ---------- bin ----------
// Inclusive tile rect; empty when x0 > x1 or y0 > y1.
fn reach(s: Shape) -> vec4i {
  let mx = vec2i(i32(U.p.tilesX) - 1, i32(U.p.tilesY) - 1);
  let whole = vec4i(0, 0, mx.x, mx.y);
  if (U.p.brute != 0u) { return whole; }
  let b = bound(s);
  let L = U.env.sunDir;
  if (b.w < 0.0 || L.y < 0.05) { return whole; }
  let r = b.w + REACH_GROW;
  let lo = b.xyz - vec3f(r);
  let hi = b.xyz + vec3f(r);
  var mn = vec2f(1e30);
  var mxp = vec2f(-1e30);
  var ok = true;
  for (var i = 0u; i < 8u; i++) {
    let c = vec3f(select(lo.x, hi.x, (i & 1u) != 0u), select(lo.y, hi.y, (i & 2u) != 0u), select(lo.z, hi.z, (i & 4u) != 0u));
    let onFloor = c - L * ((c.y - SWEEP_FLOOR) / L.y); // the corner swept along -sun to below the floor
    let a = project(c, &ok);
    let f = project(onFloor, &ok);
    mn = min(mn, min(a, f));
    mxp = max(mxp, max(a, f));
  }
  if (!ok) { return whole; }
  let t = f32(TILE);
  let lim = vec2f(f32(U.p.width), f32(U.p.height)) + t;
  let l = clamp(mn - PAD_PX, vec2f(-t), lim);
  let h = clamp(mxp + PAD_PX, vec2f(-t), lim);
  return vec4i(max(vec2i(floor(l / t)), vec2i(0)), min(vec2i(floor(h / t)), mx));
}

fn markMask(rc: vec4i, idx: u32) {
  for (var y = rc.y; y <= rc.w; y++) {
    for (var x = rc.x; x <= rc.z; x++) {
      let tile = u32(y) * U.p.tilesX + u32(x);
      atomicOr(&tileMasks[tile * MASK_WORDS + idx / 32u], 1u << (idx % 32u));
    }
  }
}

fn markDirty(rc: vec4i) {
  for (var y = rc.y; y <= rc.w; y++) {
    for (var x = rc.x; x <= rc.z; x++) {
      let tile = u32(y) * U.p.tilesX + u32(x);
      atomicOr(&dirty[tile / 32u], 1u << (tile % 32u));
    }
  }
}

fn same(a: Shape, b: Shape) -> bool {
  return all(a.pos == b.pos) && a.kind == b.kind && all(a.size == b.size) && a.round == b.round
      && all(a.rot == b.rot) && a.color == b.color && a.material == b.material && a.group == b.group
      && a.emission == b.emission;
}

@compute @workgroup_size(64)
fn bin(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  let live = i < U.p.count && shapes[i].kind != KIND_NONE;
  let wasLive = i < U.p.prevCount && prevShapes[i].kind != KIND_NONE;
  if (live) { markMask(reach(shapes[i]), i); }
  if (U.p.forceAll != 0u) { return; }
  if (live && wasLive && same(shapes[i], prevShapes[i])) { return; }
  if (live) { markDirty(reach(shapes[i])); }
  if (wasLive) { markDirty(reach(prevShapes[i])); }
}

// ---------- shade ----------
var<workgroup> wordStart: array<u32, MASK_WORDS>;
var<workgroup> list: array<u32, MAX_SHAPES>;
var<workgroup> listLen: u32;
var<workgroup> tileDirty: u32;

struct Near { d: f32, id: u32 }

fn nearest(p: vec3f, n: u32) -> Near {
  var best = Near(1e30, 0xffffffffu);
  for (var k = 0u; k < n; k++) {
    let i = list[k];
    let d = sdShape(p, shapes[i]);
    if (d < best.d) { best = Near(d, i); }
  }
  return best;
}

struct Trace { t: f32, id: u32, hit: bool }

fn march(r: Ray, n: u32) -> Trace {
  var t = 0.0;
  for (var s = 0u; s < MAX_STEPS; s++) {
    let h = nearest(r.o + r.d * t, n);
    if (h.d < 0.0005 * t + 0.02) { return Trace(t, h.id, true); }
    t += min(h.d, STEP_CAP);
    if (t > MAX_DIST) { break; }
  }
  return Trace(t, 0u, false);
}

fn normalAt(p: vec3f, n: u32) -> vec3f {
  let e = vec2f(1.0, -1.0) * 0.02;
  return normalize(e.xyy * nearest(p + e.xyy, n).d + e.yyx * nearest(p + e.yyx, n).d
                 + e.yxy * nearest(p + e.yxy, n).d + e.xxx * nearest(p + e.xxx, n).d);
}

fn softShadow(p: vec3f, L: vec3f, n: u32) -> f32 {
  let k = 1.0 / max(U.env.sunAngle, 1e-3);
  var res = 1.0;
  var t = 0.5;
  for (var s = 0u; s < SHADOW_STEPS; s++) {
    let d = min(nearest(p + L * t, n).d, STEP_CAP);
    res = min(res, k * d / t);
    if (res < 0.001 || t > SHADOW_MAX) { break; }
    t += clamp(d, 0.5, STEP_CAP);
  }
  return clamp(res, 0.0, 1.0);
}

fn ambientOcclusion(p: vec3f, nrm: vec3f, n: u32) -> f32 {
  var occ = 0.0;
  var w = 1.0;
  for (var i = 1u; i <= 4u; i++) {
    let h = AO_MAX * f32(i) / 4.0;
    let d = min(nearest(p + nrm * h, n).d, STEP_CAP);
    occ += max(h - d, 0.0) * w;
    w *= 0.6;
  }
  return clamp(1.0 - occ / AO_MAX, 0.0, 1.0);
}

fn sky(d: vec3f) -> vec3f {
  return U.env.skyColor * U.env.skyIntensity * (0.3 + 0.7 * max(d.y, 0.0));
}

fn shadePixel(pix: vec2u, n: u32) -> vec3f {
  let r = pixelRay(pix);
  let tr = march(r, n);
  if (!tr.hit) { return sky(r.d); }
  let p = r.o + r.d * tr.t;
  let s = shapes[tr.id];
  let nrm = normalAt(p, n);
  let base = pow(unpack4x8unorm(s.color).rgb, vec3f(2.2));
  let L = U.env.sunDir;
  let ndl = max(dot(nrm, L), 0.0);
  var sh = 0.0;
  if (ndl > 0.0) { sh = softShadow(p + nrm * SURF_EPS, L, n); }
  let ao = ambientOcclusion(p, nrm, n);
  let lit = U.env.sunColor * U.env.sunIntensity * ndl * sh + U.env.skyColor * U.env.skyIntensity * ao;
  return base * lit + base * s.emission;
}

@compute @workgroup_size(16, 16)
fn shade(
  @builtin(workgroup_id) wid: vec3u,
  @builtin(local_invocation_index) li: u32,
  @builtin(global_invocation_id) gid: vec3u,
) {
  let tile = wid.y * U.p.tilesX + wid.x;
  if (li == 0u) { tileDirty = (atomicLoad(&dirty[tile / 32u]) >> (tile % 32u)) & 1u; }
  if (workgroupUniformLoad(&tileDirty) == 0u) { return; }

  if (li < MASK_WORDS) { wordStart[li] = countOneBits(atomicLoad(&tileMasks[tile * MASK_WORDS + li])); }
  workgroupBarrier();
  if (li == 0u) {
    var s = 0u;
    for (var w = 0u; w < MASK_WORDS; w++) { let c = wordStart[w]; wordStart[w] = s; s += c; }
    listLen = s;
  }
  workgroupBarrier();
  if (li < MASK_WORDS) {
    var bits = atomicLoad(&tileMasks[tile * MASK_WORDS + li]);
    var o = wordStart[li];
    while (bits != 0u) {
      list[o] = li * 32u + firstTrailingBit(bits);
      o++;
      bits &= bits - 1u;
    }
  }
  let n = workgroupUniformLoad(&listLen);
  if (gid.x >= U.p.width || gid.y >= U.p.height) { return; }
  textureStore(hdr, gid.xy, vec4f(shadePixel(gid.xy, n), 1.0));
}
```

```ts
import kerfSrc from './kerf2.wgsl?raw';
import presentSrc from './present.wgsl?raw';
import { MAX_SHAPES, SHAPE_BYTES, ShapeList } from './shapes';

export const TILE = 16;
const MASK_WORDS = MAX_SHAPES / 32;
const U_BYTES = 144;

export type Vec3 = readonly [number, number, number];
const sub = (a: Vec3, b: Vec3): Vec3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const dot = (a: Vec3, b: Vec3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const cross = (a: Vec3, b: Vec3): Vec3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const norm = (a: Vec3): Vec3 => { const l = Math.hypot(a[0], a[1], a[2]); return [a[0] / l, a[1] / l, a[2] / l]; };

interface Cam { eye: Vec3; mode: 0 | 1; right: Vec3; sx: number; up: Vec3; sy: number; fwd: Vec3; near: number }

export class Kerf {
  readonly shapes = new ShapeList();
  exposure = 1;
  readonly tilesX: number;
  readonly tilesY: number;
  readonly hdr: GPUTexture;
  private u = new ArrayBuffer(U_BYTES);
  private uf = new Float32Array(this.u);
  private uu = new Uint32Array(this.u);
  private cam!: Cam;
  private sun = { dir: norm([0.35, 1, -0.45]) as Vec3, angle: 0.04, color: [1, 0.95, 0.88] as Vec3, intensity: 3 };
  private skyEnv = { color: [0.22, 0.27, 0.42] as Vec3, intensity: 0.5 };
  private envDirty = true;
  private prevCount = 0;
  private uBuf: GPUBuffer;
  private shapeBuf: GPUBuffer;
  private prevBuf: GPUBuffer;
  private maskBuf: GPUBuffer;
  private dirtyBuf: GPUBuffer;
  private expBuf: GPUBuffer;
  private bindGroup: GPUBindGroup;
  private clearPipe: GPUComputePipeline;
  private binPipe: GPUComputePipeline;
  private shadePipe: GPUComputePipeline;
  private presentPipe: GPURenderPipeline;
  private presentGroup: GPUBindGroup;

  constructor(readonly device: GPUDevice, readonly width: number, readonly height: number, readonly brute = false) {
    this.tilesX = Math.ceil(width / TILE);
    this.tilesY = Math.ceil(height / TILE);
    const tiles = this.tilesX * this.tilesY;
    const B = GPUBufferUsage;
    this.uBuf = device.createBuffer({ size: U_BYTES, usage: B.UNIFORM | B.COPY_DST });
    this.shapeBuf = device.createBuffer({ size: MAX_SHAPES * SHAPE_BYTES, usage: B.STORAGE | B.COPY_DST | B.COPY_SRC });
    this.prevBuf = device.createBuffer({ size: MAX_SHAPES * SHAPE_BYTES, usage: B.STORAGE | B.COPY_DST });
    this.maskBuf = device.createBuffer({ size: tiles * MASK_WORDS * 4, usage: B.STORAGE });
    this.dirtyBuf = device.createBuffer({ size: Math.ceil(tiles / 32) * 4, usage: B.STORAGE });
    this.expBuf = device.createBuffer({ size: 16, usage: B.UNIFORM | B.COPY_DST });
    this.hdr = device.createTexture({
      size: [width, height],
      format: 'rgba16float',
      usage: GPUTextureUsage.STORAGE_BINDING | GPUTextureUsage.TEXTURE_BINDING | GPUTextureUsage.COPY_SRC,
    });

    const C = GPUShaderStage.COMPUTE;
    const layout = device.createBindGroupLayout({
      entries: [
        { binding: 0, visibility: C, buffer: { type: 'uniform' } },
        { binding: 1, visibility: C, buffer: { type: 'read-only-storage' } },
        { binding: 2, visibility: C, buffer: { type: 'read-only-storage' } },
        { binding: 3, visibility: C, buffer: { type: 'storage' } },
        { binding: 4, visibility: C, buffer: { type: 'storage' } },
        { binding: 5, visibility: C, storageTexture: { access: 'write-only', format: 'rgba16float' } },
      ],
    });
    this.bindGroup = device.createBindGroup({
      layout,
      entries: [
        { binding: 0, resource: { buffer: this.uBuf } },
        { binding: 1, resource: { buffer: this.shapeBuf } },
        { binding: 2, resource: { buffer: this.prevBuf } },
        { binding: 3, resource: { buffer: this.maskBuf } },
        { binding: 4, resource: { buffer: this.dirtyBuf } },
        { binding: 5, resource: this.hdr.createView() },
      ],
    });
    const module = device.createShaderModule({ code: kerfSrc });
    const pl = device.createPipelineLayout({ bindGroupLayouts: [layout] });
    const compute = (entryPoint: string) => device.createComputePipeline({ layout: pl, compute: { module, entryPoint } });
    this.clearPipe = compute('clear');
    this.binPipe = compute('bin');
    this.shadePipe = compute('shade');

    const pm = device.createShaderModule({ code: presentSrc });
    this.presentPipe = device.createRenderPipeline({
      layout: 'auto',
      vertex: { module: pm, entryPoint: 'vs' },
      fragment: { module: pm, entryPoint: 'fs', targets: [{ format: navigator.gpu.getPreferredCanvasFormat() }] },
    });
    this.presentGroup = device.createBindGroup({
      layout: this.presentPipe.getBindGroupLayout(0),
      entries: [{ binding: 0, resource: this.hdr.createView() }, { binding: 1, resource: { buffer: this.expBuf } }],
    });

    this.perspective([800, 1100, 1650], [800, 0, 430], 40);
  }

  perspective(eye: Vec3, target: Vec3, fovYDeg: number): void {
    const fwd = norm(sub(target, eye));
    const side = cross(fwd, [0, 1, 0]);
    if (Math.hypot(...side) < 1e-6) throw new Error('Kerf: perspective camera cannot look straight up or down; use ortho');
    const right = norm(side);
    const up = cross(right, fwd);
    const sy = Math.tan((fovYDeg * Math.PI) / 360);
    this.cam = { eye, mode: 0, right, sx: (sy * this.width) / this.height, up, sy, fwd, near: 0.1 };
    this.envDirty = true;
  }

  // Straight down; world +z points down the screen so 2D (x, y) maps to (x, 0, y).
  ortho(cx: number, cz: number, viewW: number, height = 4000): void {
    const sx = viewW / 2;
    this.cam = { eye: [cx, height, cz], mode: 1, right: [1, 0, 0], sx, up: [0, 0, -1], sy: (sx * this.height) / this.width, fwd: [0, -1, 0], near: 0 };
    this.envDirty = true;
  }

  setSun(s: { dir: Vec3; angle: number; color: Vec3; intensity: number }): void {
    this.sun = { ...s, dir: norm(s.dir) };
    this.envDirty = true;
  }

  setSky(s: { color: Vec3; intensity: number }): void {
    this.skyEnv = { ...s };
    this.envDirty = true;
  }

  // Mirrors `project` in kerf2.wgsl. Pixel coordinates, or null if behind the camera.
  project(p: Vec3): [number, number] | null {
    const c = this.cam;
    const v = sub(p, c.eye);
    const z = dot(v, c.fwd);
    let nx: number, ny: number;
    if (c.mode === 0) {
      if (z < c.near) return null;
      nx = dot(v, c.right) / (z * c.sx);
      ny = dot(v, c.up) / (z * c.sy);
    } else {
      if (z < 0) return null;
      nx = dot(v, c.right) / c.sx;
      ny = dot(v, c.up) / c.sy;
    }
    return [((nx + 1) / 2) * this.width, ((1 - ny) / 2) * this.height];
  }

  frame(): void {
    const f = this.uf, u = this.uu, c = this.cam, s = this.sun, k = this.skyEnv;
    f.set(c.eye, 0); u[3] = c.mode; f.set(c.right, 4); f[7] = c.sx; f.set(c.up, 8); f[11] = c.sy; f.set(c.fwd, 12); f[15] = c.near;
    f.set(s.dir, 16); f[19] = s.angle; f.set(s.color, 20); f[23] = s.intensity; f.set(k.color, 24); f[27] = k.intensity;
    const count = this.shapes.count;
    u[28] = count; u[29] = this.prevCount; u[30] = this.tilesX; u[31] = this.tilesY;
    u[32] = this.brute || this.envDirty ? 1 : 0; u[33] = this.brute ? 1 : 0; u[34] = this.width; u[35] = this.height;

    const q = this.device.queue;
    q.writeBuffer(this.uBuf, 0, this.u);
    if (count) q.writeBuffer(this.shapeBuf, 0, this.shapes.data, 0, count * SHAPE_BYTES);
    const enc = this.device.createCommandEncoder();
    const pass = enc.beginComputePass();
    pass.setBindGroup(0, this.bindGroup);
    pass.setPipeline(this.clearPipe);
    pass.dispatchWorkgroups(Math.ceil((this.tilesX * this.tilesY * MASK_WORDS) / 256));
    const binN = Math.max(count, this.prevCount);
    if (binN) {
      pass.setPipeline(this.binPipe);
      pass.dispatchWorkgroups(Math.ceil(binN / 64));
    }
    pass.setPipeline(this.shadePipe);
    pass.dispatchWorkgroups(this.tilesX, this.tilesY);
    pass.end();
    if (count) enc.copyBufferToBuffer(this.shapeBuf, 0, this.prevBuf, 0, count * SHAPE_BYTES);
    q.submit([enc.finish()]);
    this.prevCount = count;
    this.envDirty = false;
  }

  async readback(): Promise<Uint8Array> {
    const row = this.width * 8;
    const stride = Math.ceil(row / 256) * 256;
    const buf = this.device.createBuffer({ size: stride * this.height, usage: GPUBufferUsage.COPY_DST | GPUBufferUsage.MAP_READ });
    try {
      const enc = this.device.createCommandEncoder();
      enc.copyTextureToBuffer({ texture: this.hdr }, { buffer: buf, bytesPerRow: stride }, [this.width, this.height]);
      this.device.queue.submit([enc.finish()]);
      await buf.mapAsync(GPUMapMode.READ);
      const src = new Uint8Array(buf.getMappedRange());
      const out = new Uint8Array(row * this.height);
      for (let y = 0; y < this.height; y++) out.set(src.subarray(y * stride, y * stride + row), y * row);
      return out;
    } finally {
      buf.destroy();
    }
  }

  present(ctx: GPUCanvasContext): void {
    this.device.queue.writeBuffer(this.expBuf, 0, new Float32Array([this.exposure, 0, 0, 0]));
    const enc = this.device.createCommandEncoder();
    const pass = enc.beginRenderPass({
      colorAttachments: [{ view: ctx.getCurrentTexture().createView(), loadOp: 'clear', storeOp: 'store', clearValue: [0, 0, 0, 1] }],
    });
    pass.setPipeline(this.presentPipe);
    pass.setBindGroup(0, this.presentGroup);
    pass.draw(3);
    pass.end();
    this.device.queue.submit([enc.finish()]);
  }

  destroy(): void {
    for (const b of [this.uBuf, this.shapeBuf, this.prevBuf, this.maskBuf, this.dirtyBuf, this.expBuf]) b.destroy();
    this.hdr.destroy();
  }
}
```

```ts
import type { Vec3 } from '../../engine/kerf2/kerf';
import { IDENTITY, ShapeList, rgba, type Quat } from '../../engine/kerf2/shapes';
import { mulberry32 } from '../scene';

export const ARENA = { w: 1600, d: 900 };
export interface LobbyOpts { blobs: number; sparks: number }
export const BUSY: LobbyOpts = { blobs: 8, sparks: 400 };
export const CALM: LobbyOpts = { blobs: 8, sparks: 0 };
export const LOBBY_CAM: { eye: Vec3; target: Vec3; fov: number } = { eye: [800, 1100, 1650], target: [800, 0, 430], fov: 40 };

const COLORS = [
  rgba(255, 60, 90), rgba(60, 200, 255), rgba(120, 255, 90), rgba(255, 200, 40),
  rgba(200, 90, 255), rgba(255, 130, 40), rgba(40, 255, 200), rgba(255, 255, 255),
];

export const qy = (a: number): Quat => [0, Math.sin(a / 2), 0, Math.cos(a / 2)];

// Pure function of (frame, opts): deterministic for equivalence tests and the bench.
export function buildLobby(s: ShapeList, frame: number, o: LobbyOpts = BUSY): void {
  const rnd = mulberry32(2026);
  const { w, d } = ARENA;
  const wall = rgba(70, 80, 120);
  s.clear();
  s.plane(0, rgba(40, 44, 60));
  s.roundbox(w / 2, 30, 10, w / 2, 30, 10, 6, IDENTITY, wall);
  s.roundbox(w / 2, 30, d - 10, w / 2, 30, 10, 6, IDENTITY, wall);
  s.roundbox(10, 30, d / 2, 10, 30, d / 2, 6, IDENTITY, wall);
  s.roundbox(w - 10, 30, d / 2, 10, 30, d / 2, 6, IDENTITY, wall);

  for (let i = 0; i < o.blobs; i++) {
    const a = frame * 0.015 + (i * Math.PI * 2) / o.blobs;
    const r = 280 + 60 * Math.sin(frame * 0.02 + i);
    const squash = 1 + 0.15 * Math.sin(frame * 0.2 + i);
    const ry = 40 / squash;
    s.ellipsoid(w / 2 + Math.cos(a) * r, ry, d / 2 + Math.sin(a) * r * 0.8, 40 * Math.sqrt(squash), ry, 40 * Math.sqrt(squash), qy(a), COLORS[i % 8], { emission: 0.6 });
  }

  for (let i = 0; i < o.sparks; i++) {
    const x0 = 60 + rnd() * (w - 120), z0 = 60 + rnd() * (d - 120), h = 20 + rnd() * 180, ph = rnd() * Math.PI * 2;
    s.sphere(x0 + Math.cos(frame * 0.03 + ph) * 25, h + Math.sin(frame * 0.05 + ph) * 10, z0 + Math.sin(frame * 0.03 + ph) * 25, 3, COLORS[i % 8], { emission: 4 });
  }
}
```

`engine/kerf2/equivalence.gpu.test.ts`:
```ts
import { afterAll, beforeAll, it } from 'vitest';
import { BUSY, CALM, LOBBY_CAM, buildLobby, qy, type LobbyOpts } from '../../bench/v2/scene';
import { getDevice } from '../kerf/gpu';
import { Kerf, type Vec3 } from './kerf';
import { IDENTITY, rgba, type ShapeList } from './shapes';

let device: GPUDevice;
beforeAll(async () => { device = await getDevice(); });
afterAll(() => device.destroy());

const W = 640, H = 360;

interface Frame { fill: (s: ShapeList) => void; setup?: (k: Kerf) => void }

function diffCount(a: Uint8Array, b: Uint8Array): number {
  let n = 0;
  for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) n++;
  return n;
}

async function expectEquivalent(frames: Frame[], width = W, height = H) {
  const fast = new Kerf(device, width, height, false);
  const ref = new Kerf(device, width, height, true);
  try {
    for (let f = 0; f < frames.length; f++) {
      for (const k of [fast, ref]) {
        frames[f].setup?.(k);
        frames[f].fill(k.shapes);
        k.frame();
      }
      const [a, b] = await Promise.all([fast.readback(), ref.readback()]);
      const d = diffCount(a, b);
      if (d) throw new Error(`frame ${f}: ${d} bytes differ`);
    }
  } finally {
    fast.destroy();
    ref.destroy();
  }
}

const lobby = (o: LobbyOpts, n: number): Frame[] =>
  Array.from({ length: n }, (_, f) => ({ fill: (s) => buildLobby(s, f, o), setup: (k) => f === 0 && k.perspective(LOBBY_CAM.eye, LOBBY_CAM.target, LOBBY_CAM.fov) }));

it('calm lobby: 8 moving blobs with sun shadows, 24 frames', async () => {
  await expectEquivalent(lobby(CALM, 24));
});

it('busy lobby: blobs + 120 sparks, 12 frames', async () => {
  await expectEquivalent(lobby({ blobs: 8, sparks: 120 }, 12));
});

it('removed shapes leave no ghost (count shrink, index shift, zero)', async () => {
  const blob = (s: ShapeList, x: number) => s.ellipsoid(x, 40, 450, 60, 40, 40, IDENTITY, rgba(255, 60, 90), { emission: 0.5 });
  await expectEquivalent([
    { fill: (s) => { s.clear(); s.plane(0, rgba(40, 44, 60)); blob(s, 600); blob(s, 1000); } },
    { fill: (s) => { s.clear(); s.plane(0, rgba(40, 44, 60)); blob(s, 1000); } },
    { fill: (s) => { s.clear(); s.plane(0, rgba(40, 44, 60)); } },
    { fill: (s) => { s.clear(); } },
  ]);
});

it('tall pillar under a low sun: long shadow far from the pillar redraws as it moves', async () => {
  const frames: Frame[] = Array.from({ length: 10 }, (_, f) => ({
    setup: (k) => f === 0 && k.setSun({ dir: [1, 0.25, 0.2], angle: 0.03, color: [1, 1, 1], intensity: 3 }),
    fill: (s) => { s.clear(); s.plane(0, rgba(90, 90, 90)); s.cylinder(500 + f * 6, 200, 450, 20, 200, 4, IDENTITY, rgba(200, 200, 200)); },
  }));
  await expectEquivalent(frames);
});

it('long thin rotating ellipsoid near tile edges (lower-bound distance) stays exact', async () => {
  const frames: Frame[] = Array.from({ length: 10 }, (_, f) => ({
    fill: (s) => { s.clear(); s.plane(0, rgba(60, 60, 70)); s.ellipsoid(800, 30, 450, 400, 12, 12, qy(f * 0.17), rgba(60, 200, 255), { emission: 1 }); },
  }));
  await expectEquivalent(frames);
});

it('shapes straddling and behind the near plane', async () => {
  const eye: Vec3 = [800, 300, 1100];
  const frames: Frame[] = Array.from({ length: 6 }, (_, f) => ({
    setup: (k) => f === 0 && k.perspective(eye, [800, 0, 400], 50),
    fill: (s) => {
      s.clear();
      s.plane(0, rgba(60, 60, 70));
      s.sphere(800 + f * 3, 300, 1100, 40, rgba(255, 200, 40), { emission: 0.5 }); // camera inside its bounds
      s.sphere(800, 300, 1300, 40, rgba(255, 0, 0)); // fully behind the camera
      s.sphere(700 + f * 10, 40, 500, 40, rgba(120, 255, 90));
    },
  }));
  await expectEquivalent(frames);
});

it('sun and camera changes redraw everything', async () => {
  const fill = (s: ShapeList) => buildLobby(s, 0, CALM);
  await expectEquivalent([
    { fill, setup: (k) => k.perspective(LOBBY_CAM.eye, LOBBY_CAM.target, LOBBY_CAM.fov) },
    { fill, setup: (k) => k.setSun({ dir: [-0.5, 1, 0.3], angle: 0.06, color: [1, 0.8, 0.7], intensity: 2 }) },
    { fill, setup: (k) => k.setSky({ color: [0.5, 0.2, 0.3], intensity: 0.8 }) },
    { fill, setup: (k) => k.perspective([600, 900, 1500], [800, 0, 450], 45) },
    { fill, setup: (k) => k.ortho(800, 450, 1600) },
  ]);
});

it('1366x768 partial tiles: fast == brute', async () => {
  await expectEquivalent(lobby(CALM, 4).map((fr, f) => ({ ...fr, setup: (k: Kerf) => f === 0 && k.perspective(LOBBY_CAM.eye, LOBBY_CAM.target, LOBBY_CAM.fov) })), 1366, 768);
});
```

`bench/v2/main.ts`:
```ts
import { getDevice } from '../../engine/kerf/gpu';
import { Kerf } from '../../engine/kerf2/kerf';
import { percentile } from '../stats';
import { BUSY, CALM, LOBBY_CAM, buildLobby, type LobbyOpts } from './scene';

const BUDGET_MS = 16.6;
const BATCH = 10; // same throughput method as the head-to-head spike: sync once per batch
const q = new URLSearchParams(location.search);
const W = Number(q.get('w') ?? 1920);
const H = Number(q.get('h') ?? 1080);
const out = document.getElementById('out')!;

async function run(device: GPUDevice, ctx: GPUCanvasContext, brute: boolean, opts: LobbyOpts, frames: number) {
  const k = new Kerf(device, W, H, brute);
  k.perspective(LOBBY_CAM.eye, LOBBY_CAM.target, LOBBY_CAM.fov);
  const times: number[] = [];
  const warm = 3 * BATCH;
  for (let f = 0; f < frames + warm; f += BATCH) {
    const t0 = performance.now();
    for (let i = 0; i < BATCH; i++) {
      buildLobby(k.shapes, f + i, opts);
      k.frame();
      k.present(ctx);
    }
    await device.queue.onSubmittedWorkDone();
    if (f >= warm) times.push((performance.now() - t0) / BATCH);
  }
  k.destroy();
  return { p50: percentile(times, 50), p99: percentile(times, 99) };
}

async function main() {
  const canvas = document.querySelector('canvas')!;
  canvas.width = W;
  canvas.height = H;
  let device: GPUDevice;
  try {
    device = await getDevice();
  } catch (e) {
    out.textContent = (e as Error).message;
    (window as unknown as { __bench2: unknown }).__bench2 = { error: (e as Error).message };
    return;
  }
  const ctx = canvas.getContext('webgpu')!;
  ctx.configure({ device, format: navigator.gpu.getPreferredCanvasFormat() });
  const rows: { scene: string; mode: string; p50: number; p99: number }[] = [];
  for (const [scene, opts] of [['busy', BUSY], ['calm', CALM]] as const) {
    rows.push({ scene, mode: 'kerf2', ...(await run(device, ctx, false, opts, 600)) });
    rows.push({ scene, mode: 'brute', ...(await run(device, ctx, true, opts, 20)) });
  }
  const pass = rows.filter((r) => r.mode === 'kerf2').every((r) => r.p99 <= BUDGET_MS);
  out.textContent = `${W}x${H}\n` + rows.map((r) => `${r.scene.padEnd(5)} ${r.mode.padEnd(6)} p50 ${r.p50.toFixed(2)}ms  p99 ${r.p99.toFixed(2)}ms`).join('\n') + `\n${pass ? 'PASS' : 'FAIL'} (p99 ≤ ${BUDGET_MS}ms)`;
  (window as unknown as { __bench2: unknown }).__bench2 = { width: W, height: H, rows, pass };
}

main();
```

`bench/v2/run.mjs`:
```js
// Usage: npm run dev (another terminal), then: npm run bench:v2
import { chromium } from 'playwright';

const url = process.argv[2] ?? 'http://localhost:5173/bench/v2/';
const browser = await chromium.launch({ channel: 'chrome', headless: false, args: ['--enable-unsafe-webgpu'] });
let allPass = true;
for (const throttle of [1, 4])
  for (const [w, h] of [[1920, 1080], [3840, 2160]]) {
    const page = await browser.newPage({ viewport: { width: 1280, height: 720 } });
    const cdp = await page.context().newCDPSession(page);
    await cdp.send('Emulation.setCPUThrottlingRate', { rate: throttle });
    await page.goto(`${url}?w=${w}&h=${h}`);
    const r = await (await page.waitForFunction(() => window.__bench2, null, { timeout: 20 * 60_000 })).jsonValue();
    await page.close();
    if (r.error) { console.error('FAIL:', r.error); process.exit(1); }
    console.log(`\n${w}x${h} CPU ${throttle}x — ${r.pass ? 'PASS' : 'FAIL'}`);
    console.table(r.rows.map((x) => ({ ...x, p50: +x.p50.toFixed(2), p99: +x.p99.toFixed(2) })));
    allPass &&= r.pass;
  }
await browser.close();
process.exit(allPass ? 0 : 1);
```
